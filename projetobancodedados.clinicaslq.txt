CREATE TABLE medicos (
    crm INT PRIMARY KEY,
    nome VARCHAR(28),
    idade INT,
    email VARCHAR(30),
    telefone VARCHAR(30),
    especializacao VARCHAR(28)
    CHECK (especializacao IN ('cadiologista', 'nutricionista'))
);


CREATE TABLE pacientes (
    id_paciente INT PRIMARY KEY,
    nome VARCHAR(28),
    idade INT,
    email VARCHAR(30),
    telefone VARCHAR(30),
    endereco VARCHAR(30),
    tipo_exame VARCHAR(20)
    CHECK (tipo_exame IN ('clinico', 'laboratorial'))
);


CREATE TABLE consultas (
    id_medico INT NOT NULL,
    id_paciente INT NOT NULL,
    data_consulta DATE,
    horario_consulta TIME,
    valor_consulta DOUBLE NOT NULL,
    tipo_pagamento VARCHAR(20),
    FOREIGN KEY (id_medico) REFERENCES medicos(crm),
    FOREIGN KEY (id_paciente) REFERENCES pacientes(id_paciente)
);


CREATE TABLE exame (
    id_medico INT NOT NULL,
    id_paciente INT NOT NULL,
    data_exame DATE,
    horario TIME,
    valor_exame DOUBLE NOT NULL,
    tipo_pagamento_exame VARCHAR(28) UNIQUE,
    tipo_exame VARCHAR(50) UNIQUE,

    FOREIGN KEY (id_medico) REFERENCES medicos(crm),
    FOREIGN KEY (id_paciente) REFERENCES pacientes(id_paciente)
);
